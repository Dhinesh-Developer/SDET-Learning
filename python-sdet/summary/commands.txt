command                     purpose
pytest -v                   detailed test output
pytest -q                   concise test output
pytest -s                   show print output
pytest tests.test_basic.py  run one file
pytest -k login             run tests matching a name
pytest --collect-only       list discovered tests


pytest -m smoke              smoke tests only
pytest -m regression         regression tests only
pytest -m "smoke or sanity"  either marker
pytest -m "not slow"         exclude slow tests
pytest -m "smoke and regression" tests with both markers
